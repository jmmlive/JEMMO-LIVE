// ─────────────────────────────────────────────────────────────
// JEMMO LIVE · Presencia pública autenticada · PRUEBA 54
// ─────────────────────────────────────────────────────────────
match /livePresences/{hostUid} {
  function signedIn() {
    return request.auth != null;
  }

  function isHost() {
    return signedIn() && request.auth.uid == hostUid;
  }

  allow get, list: if signedIn();
  allow create, update: if isHost()
    && request.resource.data.uid == hostUid
    && request.resource.data.hostUid == hostUid;
  allow delete: if isHost();
}

// ─────────────────────────────────────────────────────────────
// JEMMO LIVE · Señalización WebRTC autenticada · PRUEBA 54

match /liveSignals/{hostUid} {
  function signedIn() {
    return request.auth != null;
  }

  function isHost() {
    return signedIn() && request.auth.uid == hostUid;
  }

  // La sala se identifica exclusivamente por el UID autenticado del anfitrión.
  allow get: if signedIn();
  allow list: if false;
  allow create, update: if isHost()
    && request.resource.data.hostUid == hostUid
    && request.resource.data.roomSessionId is string
    && request.resource.data.roomSessionId.size() >= 12;
  allow delete: if isHost();

  // Chat LIVE sincronizado por sesión. Los mensajes son inmutables.
  match /chat/{messageId} {
    function signedIn() {
      return request.auth != null;
    }

    function isHost() {
      return signedIn() && request.auth.uid == hostUid;
    }

    function isSender() {
      return signedIn() && resource.data.senderUid == request.auth.uid;
    }

    function activeRoomSession() {
      return exists(/databases/$(database)/documents/liveSignals/$(hostUid))
        && get(/databases/$(database)/documents/liveSignals/$(hostUid)).data.active == true
        && get(/databases/$(database)/documents/liveSignals/$(hostUid)).data.status == 'live'
        && get(/databases/$(database)/documents/liveSignals/$(hostUid)).data.roomSessionId == request.resource.data.roomSessionId;
    }

    allow get, list: if signedIn();

    allow create: if signedIn()
      && request.resource.data.hostUid == hostUid
      && request.resource.data.senderUid == request.auth.uid
      && request.resource.data.roomSessionId is string
      && request.resource.data.roomSessionId.size() >= 12
      && request.resource.data.senderName is string
      && request.resource.data.senderName.size() >= 1
      && request.resource.data.senderName.size() <= 80
      && request.resource.data.text is string
      && request.resource.data.text.size() >= 1
      && request.resource.data.text.size() <= 300
      && request.resource.data.type in ['chat', 'sticker', 'gift', 'system']
      && request.resource.data.senderRole in ['host', 'viewer']
      && (request.resource.data.senderRole != 'host' || request.auth.uid == hostUid)
      && (request.resource.data.type != 'system' || request.auth.uid == hostUid)
      && request.resource.data.keys().hasOnly([
        'hostUid', 'roomSessionId', 'senderUid', 'senderName',
        'senderRole', 'type', 'text', 'createdAt', 'createdAtMs', 'version'
      ])
      && activeRoomSession();

    allow update: if false;
    allow delete: if isHost() || isSender();
  }

  match /viewers/{sessionId} {
    function signedIn() {
      return request.auth != null;
    }

    function isHost() {
      return signedIn() && request.auth.uid == hostUid;
    }

    function isViewerExisting() {
      return signedIn() && resource.data.viewerUid == request.auth.uid;
    }

    function activeRoomSession() {
      return exists(/databases/$(database)/documents/liveSignals/$(hostUid))
        && get(/databases/$(database)/documents/liveSignals/$(hostUid)).data.active == true
        && get(/databases/$(database)/documents/liveSignals/$(hostUid)).data.status == 'live'
        && get(/databases/$(database)/documents/liveSignals/$(hostUid)).data.roomSessionId == request.resource.data.roomSessionId;
    }

    function stableOwnership() {
      return request.resource.data.viewerUid == resource.data.viewerUid
        && request.resource.data.hostUid == resource.data.hostUid
        && request.resource.data.hostUid == hostUid
        && request.resource.data.roomSessionId == resource.data.roomSessionId;
    }

    function viewerCreateFields() {
      return request.resource.data.keys().hasOnly([
        'viewerUid', 'hostUid', 'roomSessionId', 'status',
        'offerRevision', 'answerRevision',
        'createdAt', 'createdAtMs', 'updatedAt', 'updatedAtMs',
        'userAgent', 'protocol', 'turnConfigured', 'version'
      ]);
    }

    function viewerUpdateFields() {
      return request.resource.data.diff(resource.data).affectedKeys().hasOnly([
        'offer', 'offerRevision', 'offerIceComplete', 'offerIceTimedOut',
        'offerCandidates', 'status', 'requestedAt', 'requestedAtMs',
        'updatedAt', 'updatedAtMs', 'protocol', 'version'
      ]);
    }

    function hostUpdateFields() {
      return request.resource.data.diff(resource.data).affectedKeys().hasOnly([
        'answer', 'answerRevision', 'answerIceComplete', 'answerIceTimedOut',
        'answerCandidates', 'status', 'hostUid', 'media',
        'answeredAt', 'answeredAtMs', 'connectedAt', 'connectedAtMs',
        'endedAt', 'endedAtMs', 'endReason',
        'capacity', 'errorCode', 'errorMessage',
        'updatedAt', 'updatedAtMs', 'protocol', 'version'
      ]);
    }

    // Solo el anfitrión puede listar las solicitudes de su propia sala.
    allow list: if isHost();
    allow get: if isHost() || isViewerExisting();

    // El espectador crea una sesión para el UID del anfitrión y el roomSessionId activo.
    allow create: if signedIn()
      && request.resource.data.viewerUid == request.auth.uid
      && request.resource.data.hostUid == hostUid
      && request.resource.data.roomSessionId is string
      && request.resource.data.roomSessionId.size() >= 12
      && request.resource.data.status == 'initializing'
      && request.resource.data.offerRevision == 0
      && request.resource.data.answerRevision == 0
      && activeRoomSession()
      && viewerCreateFields();

    // El espectador solo publica su oferta o cierra su propia sesión.
    allow update: if isViewerExisting()
      && stableOwnership()
      && viewerUpdateFields();

    // El anfitrión solo publica respuesta, estado y diagnóstico.
    allow update: if isHost()
      && stableOwnership()
      && hostUpdateFields();

    allow delete: if isHost() || isViewerExisting();

    match /viewerCandidates/{candidateId} {
      function signedIn() {
        return request.auth != null;
      }

      function isHost() {
        return signedIn() && request.auth.uid == hostUid;
      }

      function isSessionViewer() {
        return signedIn()
          && exists(/databases/$(database)/documents/liveSignals/$(hostUid)/viewers/$(sessionId))
          && get(/databases/$(database)/documents/liveSignals/$(hostUid)/viewers/$(sessionId)).data.viewerUid == request.auth.uid;
      }

      allow get, list: if isHost() || isSessionViewer();
      allow create: if isSessionViewer()
        && request.resource.data.ownerUid == request.auth.uid
        && request.resource.data.keys().hasOnly([
          'candidate', 'sdpMid', 'sdpMLineIndex', 'usernameFragment',
          'ownerUid', 'createdAt', 'createdAtMs', 'version'
        ]);
      allow update: if false;
      allow delete: if isHost() || isSessionViewer();
    }

    match /hostCandidates/{candidateId} {
      function signedIn() {
        return request.auth != null;
      }

      function isHost() {
        return signedIn() && request.auth.uid == hostUid;
      }

      function isSessionViewer() {
        return signedIn()
          && exists(/databases/$(database)/documents/liveSignals/$(hostUid)/viewers/$(sessionId))
          && get(/databases/$(database)/documents/liveSignals/$(hostUid)/viewers/$(sessionId)).data.viewerUid == request.auth.uid;
      }

      allow get, list: if isHost() || isSessionViewer();
      allow create: if isHost()
        && request.resource.data.ownerUid == request.auth.uid
        && request.resource.data.keys().hasOnly([
          'candidate', 'sdpMid', 'sdpMLineIndex', 'usernameFragment',
          'ownerUid', 'createdAt', 'createdAtMs', 'version'
        ]);
      allow update: if false;
      allow delete: if isHost() || isSessionViewer();
    }
  }
}
